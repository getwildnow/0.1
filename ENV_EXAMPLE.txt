# ============================================
# SUPABASE CONFIGURATION
# ============================================
# Get these from: https://supabase.com/dashboard/project/_/settings/api
# 
# NEXT_PUBLIC_SUPABASE_URL: Your Supabase project URL
#   Example: https://abcdefghijklmnop.supabase.co
#
# NEXT_PUBLIC_SUPABASE_ANON_KEY: Your Supabase anonymous/public key
#   This is safe to expose in the browser
#   Example: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
#
# SUPABASE_SERVICE_ROLE_KEY: Your Supabase service role key (server-side only!)
#   ⚠️ NEVER expose this in the browser - it bypasses RLS
#   Example: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...

NEXT_PUBLIC_SUPABASE_URL=https://your-project-id.supabase.co
NEXT_PUBLIC_SUPABASE_ANON_KEY=your_supabase_anon_key_here
SUPABASE_SERVICE_ROLE_KEY=your_supabase_service_role_key_here

# ============================================
# VERIFF CONFIGURATION (Identity Verification)
# ============================================
# Get these from: https://station.veriff.com/
#
# VERIFF_API_KEY: Your Veriff API key (server-side only!)
#   ⚠️ NEVER expose this in the browser
#   Get from: Veriff Dashboard > Settings > API Keys
#
# VERIFF_API_SECRET: Your Veriff API secret (for webhook verification)
#   ⚠️ NEVER expose this in the browser
#   Used to verify webhook signatures
#   Get from: Veriff Dashboard > Settings > API Keys
#
# VERIFF_API_URL: Veriff API base URL (optional)
#   Default: https://stationapi.veriff.com
#   Usually don't need to change this
#   Note: API endpoints include /v1/ (e.g., /v1/sessions)

VERIFF_API_KEY=your_veriff_api_key_here
VERIFF_API_SECRET=your_veriff_api_secret_here
VERIFF_API_URL=https://stationapi.veriff.com

# ============================================
# SETUP INSTRUCTIONS
# ============================================
#
# 1. SUPABASE:
#    - Go to https://supabase.com and create a project
#    - Run migrations from supabase/migrations/ folder
#    - Copy URL and keys from Settings > API
#
# 2. VERIFF:
#    - Go to https://station.veriff.com/ and create an account
#    - Get API key and secret from Settings > API Keys
#    - Set up webhook: Settings > Webhooks
#      - Endpoint: https://your-domain.com/api/webhooks/veriff
#      - Events: verification.status.changed
#      - Copy API secret for signature verification
#
# 3. OPENAI:
#    - Go to https://platform.openai.com
#    - Create API key in Settings > API keys
#    - Add billing/payment method
#
# 4. CREATE .env.local:
#    - Copy this file to .env.local
#    - Fill in all values above
#    - Restart dev server: npm run dev
#
